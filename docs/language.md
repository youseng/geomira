# The GEOMIRA language

Generated by `pnpm docs:lang` (packages/lang/src/docs.ts) from the grammar, the notation
tables, the claim kinds and the tool registry. Do not edit by hand: a test fails when this
file is stale. Decisions: docs/adr/0009-grammar.md.

## Grammar

```ebnf
(* ---- lexical ---- *)
file          = { line , newline } , [ line ] ;
line          = { space } , [ statement ] , { space } , [ comment ] ;
comment       = "#" , { any character } ;                  (* kept: it belongs to the next statement *)
space         = " " | tab ;                                 (* a trailing CR is dropped *)

pointName     = upper , [ digit , { digit }
                        | "_" , ( digit , { digit } | letter | "{" , alnum , { alnum } , "}" ) ] ,
                { "'" } , { "*" } ;                         (* glued stars always belong to it *)
otherName     = ( lower | greek ) , { letter | digit | "_" | greek } , { "'" } , [ stars ] ;
stars         = "*" , { "*" } ;     (* part of otherName iff the next character is NOT a letter,
                                       digit, "_", "(" or "\"; a starred name followed by
                                       whitespace and an operand is an error *)
greek         = "α" … "ω" | "Α" … "Ω" | "\" , greekWord ;  (* \alpha … \omega, \Gamma … \Omega *)
name          = pointName | otherName ;
number        = digit , { digit } , [ "." , digit , { digit } | "/" , digit , { digit } ] , [ "°" ] ;
                                    (* exact; glued; a leading "-" glued to it makes it negative *)
float         = [ "-" ] , digit , { digit } , [ "." , digit , { digit } ] ,
                [ ( "e" | "E" ) , [ "+" | "-" ] , digit , { digit } ] ;   (* attributes, @viewport *)
string        = '"' , { character | "\\" , ( '"' | "\\" | "n" | "t" | "r" | "u{" , hex , "}" ) } , '"' ;
symbol escapes: "\angle" = "∠", "\measuredangle" = "∡", "\cdot" = "·", "\neq" = "\ne" = "≠";
                "−" = "-", "⋅" = "·", "!=" = "≠", "<=" = "≤", ">=" = "≥".

(* ---- statements: one per line ---- *)
statement     = definition | claimStmt | assumeStmt | triangleStmt | styleStmt | labelStmt
              | directive ;
definition    = name , "=" , expr , { attrGroup } ;        (* keys: style keys, layers *)
claimStmt     = "claim" , claim , { attrGroup } ;           (* key: name *)
assumeStmt    = "assume" , condition ;
triangleStmt  = "triangle" , pointName , pointName , pointName , { attrGroup } ;
                                    (* keys: acute, scalene, obtuse_at_<vertex> *)
styleStmt     = "style" , name , { attrGroup } ;           (* style keys; images too: style A* … *)
labelStmt     = "label" , name , string ;
directive     = "@version" , digit , { digit }
              | "@title" , string | "@author" , string
              | "@layer" , name , [ "=" , "image" , "(" , name , "," , name , ")" ] , { attr }
                                    (* keys: main, visible, locked, opacity, order;
                                       derived: suffix, exclude *)
              | "@view" , name , { attr }                   (* keys: layers, viewport *)
              | "@viewport" , float , float , float ;       (* centre x, centre y, zoom *)

(* ---- attributes ---- *)
attr          = attrGroup | attrItem ;                      (* bare items in directives only *)
attrGroup     = "[" , [ attrItem , { "," , attrItem } ] , "]" ;
attrItem      = key , [ "=" , value , { "," , value } ] ;   (* a list (exclude, layers, viewport)
                                                               takes the rest of its group *)
value         = float | number | name | "-" , name | string | stars ;
style keys    : color=word|string, width=float, dash=solid|dashed|dotted, hidden[=true|false],
                noimage[=true|false], label=string, z=float, label_angle=float (degrees),
                label_dist=float (pixels, >= 0).

(* ---- expressions ---- *)
expr          = product , { ( "+" | "-" ) , product } ;
product       = unary , { ( "*" | "·" | "/" ) , unary } ;
unary         = "-" , number                                (* glued: a negative literal *)
              | "-" , unary | primary ;
primary       = number | name | call | "(" , expr , ")" | "(" , expr , "," , expr , ")" ;
                                                            (* (x, y) is point(x, y) *)
call          = tool , "(" , [ args ] , ")" ;               (* tool: a registry name *)
args          = arg , { "," , arg } , { "," , selector } | selector , { "," , selector } ;
arg           = expr | string ;                             (* on(obj) gets an explicit parameter;
                                                               a string only as a whole argument:
                                                               text(P, "…") (ADR-0011) *)
selector      = ( "same_side" | "opposite_side" ) , "(" , expr , "," , expr , ")"
              | "on_arc" , "(" , expr , "," , expr , "," , [ "not" ] , expr , ")"
              | "between" , "(" , expr , "," , expr , ")"
              | ( "inside" | "outside" ) , "(" , expr , ")"
              | ( "other" | "near" | "far_from" ) , "(" , expr , ")"     (* other(P): one argument *)
              | ( "along" | "orient" ) , "(" , ( "+" | "-" ) , ")" ;

(* ---- claims (kinds of ADR-0008) ---- *)
claim         = kind , "(" , expr , { "," , expr } , ")"    (* every kind by name *)
              | segment , "=" , [ number , ( "·" | "*" ) ] , segment          (* length *)
              | segment , "/" , segment , "=" , segment , "/" , segment      (* ratio *)
              | segment , "/" , segment , "=" , number                       (* ratio_value *)
              | [ number , [ "·" | "*" ] ] , dangle , "=" , dangle           (* dangle *)
              | "∡" , "(" , expr , "," , expr , ")" , "=" , number           (* dangle_lines *)
              | angle , "=" , angle                                          (* angle_eq *)
              | angle , ( "+" | "-" ) , angle , "=" , number                 (* angle_sum, angle_diff *)
              | "harmonic" , "(" , expr , "," , expr , ";" , expr , "," , expr , ")"
                                    (* harmonic, or harmonic_lines on four lines (ADR-0040) *)
              | "cross_ratio" , "(" , expr , "," , expr , [ "," | ";" ] , expr , "," , expr , ")" ,
                "=" , number        (* cross_ratio, or cross_ratio_lines on four lines *)
              | expr , "=" , expr ;                                          (* equal *)
segment       = pointName , pointName                       (* glued: AB *)
              | "dist" , "(" , expr , "," , expr , ")" ;
angle         = "∠" , pointName , pointName , pointName     (* glued: ∠ABC, the angle at B *)
              | "∠" , "(" , expr , "," , expr , "," , expr , ")"
              | "angle" , "(" , expr , "," , expr , "," , expr , ")" ;
dangle        = "∡" , pointName , pointName , pointName
              | "∡" , "(" , expr , "," , expr , "," , expr , ")" ;

(* ---- assumptions (open conditions) ---- *)
condition     = kind , "(" , expr , { "," , expr } , ")"
              | measure , ( "<" | ">" | "≠" ) , measure
              | expr , "inside" , expr
              | expr , { "," , expr } , "in" , "order" , "on" , expr ;
measure       = segment | angle | number | expr ;
```

## Statements

| statement | meaning |
| --- | --- |
| `name = expr [attrs]` | an object (points: `A`, `M_a`, `H'`; others: `l`, `ω`, `k2`) |
| `claim <claim> [name=…]` | a claim (see Claims) |
| `assume <condition>` | a hypothesis: an open condition (see Assumptions) |
| `triangle A B C [type]` | three free points (or the ones that exist), `tri_ABC = polygon(A, B, C)` and the type's `assume` lines; types acute, scalene, obtuse_at_<vertex> |
| `points A B C [attrs]` | free points at places drawn from the names (Geomdraw's notation) |
| `X, Y = intersect(a, b)` | both branches of a two-candidate call (Geomdraw's notation, see below) |
| `style X [attrs]` | the style of X (also of an image: `style A* [hidden]`) |
| `label X "…"` | the label of X |
| `@version 1` | the file version (always printed; current: 1) |
| `@title "…"`, `@author "…"` | document information |
| `@layer name [visible=false] [locked] [opacity=0.5] [order=n] [main]` | a layer; `[main]` names the main layer |
| `@layer inv = image(T, main) [suffix=*] [exclude=X,Y]` | a derived layer: the image under T of every object of the source layer |
| `@view name [layers=inv,-main] [viewport=cx,cy,zoom]` | a view: layers shown (name) or hidden (-name), optional viewport |
| `@viewport cx cy zoom` | the viewport |
| `# comment` | a comment: kept with the statement that follows it (or on the same line) |

## Attributes

| statement | keys |
| --- | --- |
| definition | color, width, dash, opacity, hidden, noimage, label, layers |
| style | color, width, dash, opacity, hidden, noimage, label |
| claim | name |
| triangle | acute, scalene, obtuse_at_<vertex> |
| @layer | main, visible, locked, opacity, order; derived layers also suffix, exclude |
| @view | layers, viewport |

Values: `color=red` or `color="#ff0000"`; `width=1.5`; `dash=solid|dashed|dotted`; `opacity=0.5` (in [0, 1], times the layer's opacity); flags `hidden`, `noimage`, `locked` (or `=true` / `=false`); `label="…"`; lists `exclude=X,Y`, `layers=a,b` (a list takes the rest of its bracket group).

## Selectors

Selectors are the trailing arguments of a branchable call (`intersect`, `tangent_point`, `tangent_from`). A bare index (`intersect(a, b, 2)`) and a call with two candidates and no selector are errors with a hint.

| selector | meaning |
| --- | --- |
| `same_side(Q, l)` | filter: on the side of line l where Q is |
| `opposite_side(Q, l)` | filter: on the other side of line l from Q |
| `on_arc(B, C, A) or on_arc(B, C, not A)` | filter: on the arc BC that contains A (not: the other arc) |
| `between(A, B)` | filter: strictly between the perpendiculars to AB at A and at B |
| `inside(c)` | filter: inside circle c |
| `outside(c)` | filter: outside circle c |
| `other(P)` | ranker: the intersection other than P (exact: no square root) |
| `near(P)` | ranker: the candidate nearer to P |
| `far_from(P)` | ranker: the candidate farther from P |
| `along(+) or along(-)` | low-level, line ∩ circle: further (+) or less far (-) along the line's direction |
| `orient(+) or orient(-)` | low-level, circle ∩ circle and tangent choices: the side by orientation |

## Assumptions

Open conditions only: equality assumptions are errors ("construct it"), and so are ≤ and ≥.

| kind | notation | example |
| --- | --- | --- |
| `acute` | `acute(A, B, C)` | `assume acute(A, B, C)` |
| `scalene` | `scalene(A, B, C)` | `assume scalene(A, B, C)` |
| `convex` | `convex(A, B, C, D, …)` | `assume convex(A, B, M_a, M_b)` |
| `inside` | `P inside poly`, `inside(P, poly)` | `assume H inside polygon(A, B, C)` |
| `in_order` | `A, B, C in order on curve`, `in_order(A, B, C, curve)` | `assume B, C, A in order on ω` |
| `less` | `x < y (x, y: AB, ∠ABC, angle(…), dist(…), numbers)` | `assume AB < AC` |
| `greater` | `x > y` | `assume ∠BAC > 30` |
| `not_equal` | `x ≠ y`, `x != y` | `assume AB ≠ AC` |

## Geomdraw's notation

The notation of the maintainer's earlier Geomdraw is read too (ADR-0025). It is turned into the forms above when the text is applied, so a saved file holds only those. Examples on the figure `A = (1, 2); B = (3, 4); C = (0, 5); D = (2, 7); P = (9, 9); ω = ABC`:

| written | read as |
| --- | --- |
| `points E F` | `E = (3.5, 1.7) · F = (-3.9, 2)` |
| `Q = point()` | `Q = (2, -2.1)` |
| `triangle ABC` | `tri_ABC = polygon(A, B, C)` |
| `M = midpoint(BC)` | `M = midpoint(B, C)` |
| `O = circumcenter(ABC)` | `O = circumcenter(A, B, C)` |
| `H = foot(C, AB)` | `H = foot(C, line(A, B))` |
| `X, Y = intersect(ω, AB)` | `X = intersect(ω, line(A, B), along(+)) · Y = intersect(ω, line(A, B), along(-))` |
| `t_1, t_2 = tangents(P, ω)` | `t_1 = tangent_from(P, ω, orient(+)) · t_2 = tangent_from(P, ω, orient(-))` |
| `l = AB @red @dashed` | `l = line(A, B) [color=red, dash=dashed]` |
| `G = Centroid(A, B, C)` | `G = centroid(A, B, C)` |
| `claim A, B, C collinear` | `claim collinear(A, B, C)` |
| `claim ABCD cyclic` | `claim concyclic(A, B, C, D)` |
| `claim AB ⟂ CD` | `claim perpendicular(line(A, B), line(C, D))` |
| `claim AD ∥ BC` | `claim parallel(line(A, D), line(B, C))` |
| `claim D on ω` | `claim on(D, ω)` |

Style words: `@dashed`, `@dotted`, `@solid`, `@thick`, `@thin`, `@hidden`, `@nolabel`, `@noimage`, `@black`, `@red`, `@blue`, `@green`, `@orange`, `@purple`, `@brown`, `@teal`, `@gray`, `@grey`.

Other names: `tangents` = `tangent_from`, `tangent_points` = `tangent_point`, `second_intersection` = `other`, `length` = `dist`, `distance` = `dist`, `slider` = `param`, `perpendicular` = `perp`.

```ebnf
(* ---- Geomdraw's notation: read, never printed (ADR-0025) ---- *)
statement    += pointsStmt | branchesStmt ;
pointsStmt    = ( "point" | "points" ) , pointName , { pointName } , { attr } ;
                                    (* free points at places drawn from the names *)
branchesStmt  = name , "," , name , "=" , call , { attr } ;
                                    (* a two-candidate call without selectors: the first name
                                       takes along(+) or orient(+), the second along(-) or orient(-) *)
definition   += name , "=" , "point" , "(" , ")" , { attr } ;           (* a free point *)
triangleStmt += "triangle" , pointName , pointName , pointName ,
                { "acute" | "scalene" | "obtuse" | "obtuse_at_" , pointName } , { attr } ;
                                    (* names that exist are used; "obtuse": at the first vertex *)
primary      += run ;
run           = pointName , pointName , { pointName } ;    (* glued: AB is line(A, B), ABC is
                                       circle(A, B, C), ABCD… polygon(A, B, C, D, …); as an argument
                                       a run is spread into its points where the call or the claim
                                       takes the points and not the line, circle or polygon *)
call         += Capitalised , "(" , [ args ] , ")" ;    (* Midpoint(…) is midpoint(…) *)
tool         += alias ;                                     (* see "Other names" *)
attr         += "@" , styleWord ;                           (* @red is [color=red] *)
claim        += expr , { "," , expr } , ( "collinear" | "concyclic" | "cyclic" | "concurrent" )
              | expr , ( "⟂" | "⊥" | "perp" | "perpendicular" | "∥" | "parallel"
                       | "on" | "tangent" ) , expr ;
symbol escapes += "\perp" = "⟂", "\parallel" = "∥".
keywords: Triangle, Point, Points, Claim, Assume, Style, Label start a line as in lowercase.
```
